take = async(q,batch)=>{await b3state.tab.goto(`http://127.0.0.1:${q.port}/?office-lab=${q.scenario}&motion=still&vf-metrics&round=${q.round}&batch=${batch}`);await b3state.tab.playwright.locator('#vf-metrics[data-samples="180"]').waitFor({state:'attached',timeoutMs:20000});let d=JSON.parse(await b3state.tab.playwright.locator('#vf-metrics').textContent()),s=JSON.parse(await b3state.tab.playwright.locator('#vf-startup').textContent()),row={label:q.label,mode:q.mode,scenario:q.scenario,round:q.round,batch,at:new Date().toISOString(),...d,visibility:s.visibility};if(d.raf.p50>25||s.visibility!=='visible')throw Error('environment '+JSON.stringify(row));return row;}

warmMode = async(mode)=>{for(const [label,port] of b3state.versions)for(const scenario of ['standard','twelve']){b3state.warm.push(await b3state.take({label,port,scenario,mode,round:0},'warmup'));await b3state.save('startup-warmup-excluded.json',b3state.warm);}nodeRepl.write({warmups:b3state.warm.length});}

nextPerf = async(n)=>{for(let i=0;i<n&&b3state.perfQueue.length;i++){b3state.perf.push(await b3state.take(b3state.perfQueue[0],'primary'));b3state.perfQueue.shift();await b3state.save('startup-balanced.json',b3state.perf);await b3state.save('performance-remaining.json',b3state.perfQueue);}nodeRepl.write({formal:b3state.perf.length,remaining:b3state.perfQueue.length});}
