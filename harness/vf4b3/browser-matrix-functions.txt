open = async(scenario,size)=>{if(b3state.size?.width!==size.width||b3state.size?.height!==size.height){await vp.set(size);b3state.size=size;}let url='http://127.0.0.1:5184/?office-lab='+scenario+'&motion=still'+(size.width===560?'&locale=en&theme=dark':'');if(b3state.lastURL===url)await b3state.tab.reload();else await b3state.tab.goto(url);b3state.lastURL=url;await b3state.tab.playwright.locator('#b3-render-audit[data-ready=true]').waitFor({state:'attached'});}

zone = async(z)=>{await b3state.tab.playwright.getByRole('button',{name:/^视角$|^View$/}).click();await b3state.tab.playwright.locator('.office-camera button').nth(['overview','work-left','work-right','meeting','lounge'].indexOf(z)).click();await b3state.tab.playwright.locator('.office-camera button[aria-pressed=true]').filter({hasText:z==='lounge'?/休闲区|Lounge/:z==='meeting'?/会议区|Meeting/:/工作区Ⅱ|Work II/}).waitFor({state:'visible'});await b3state.tab.playwright.getByRole('button',{name:/^视角$|^View$/}).click();await b3state.tab.playwright.getByRole('button',{name:/^返回全景$|^Reset view$/}).waitFor({state:'visible'});await b3state.tab.playwright.locator('#b3-render-audit[data-camera="'+z+'"]').waitFor({state:'attached'});}

cap = async(label,shot=false,path=null)=>{let d=JSON.parse(await b3state.tab.playwright.locator('#b3-render-audit').textContent());let ui=await b3state.tab.playwright.evaluate(()=>({resetVisible:!!document.querySelector('.reset-view'),resetText:document.querySelector('.reset-view')?.textContent,presence:document.querySelector('.scene-tools')?.textContent}));let row={label,...d,observedUi:ui};await fs.writeFile(b3dir+'records/'+String(b3state.rows.length).padStart(5,'0')+'.json',JSON.stringify(row));b3state.rows.push(row);if(shot){await b3state.tab.playwright.domSnapshot();let filename=path??b3dir+label+'.png';await fs.writeFile(filename,await b3state.tab.screenshot({fullPage:false}));b3state.shots.push({label,path:filename,scenario:d.scenario,camera:d.layoutFocus?'locate':d.zone,viewport:d.viewport,background:d.background,ui});await b3state.save('screenshot-manifest.json',b3state.shots);}return d;}

check = d=>{const ids=['bluewhale','qwen','llama','mistral','glm'],inside=p=>p.x>=0&&p.y>=0&&p.x<d.viewport[0]&&p.y<d.viewport[1],a=d.actors.filter(a=>ids.includes(a.id)),kind=d.scenario.split('-')[1];return{face:a.every(a=>a.coveredPixels===0),badges:a.every(a=>!a.badgeVisible||a.badgeOverlap===0),insideBadges:d.actors.every(a=>!a.insideFrame||a.badgeVisible),faceHit:a.every(a=>!inside(a.faceHit)||a.faceHit.actual==='actor:'+a.id),bodyHit:a.every(a=>!inside(a.bodyHit)||a.bodyHit.actual==='actor:'+a.id),transparent:d.furnitureHits.filter(f=>f.kind===kind).every(f=>f.probes.filter(p=>['transparent-corner','gap','shadow'].includes(p.name)).every(p=>!inside(p.screen)||p.actual!=='furniture:'+f.id)),solid:d.furnitureHits.filter(f=>f.kind===kind).every(f=>f.probes.filter(p=>!['transparent-corner','gap','shadow'].includes(p.name)).every(p=>!inside(p.screen)||p.actual==='furniture:'+f.id||p.actual?.startsWith('actor:')))}}

next = async(n)=>{for(let i=0;i<n&&b3state.queue.length;i++){const q=b3state.queue[0],stem='final-'+q.scenario.slice(3)+'-'+q.size.width;await b3state.open(q.scenario,q.size);let a=await b3state.cap(stem+'-overview',q.shot);await b3state.zone(q.zone);let b=await b3state.cap(stem+'-zone',q.shot);for(const d of [a,b]){let c=b3state.check(d);if(Object.values(c).some(v=>!v))throw Error(JSON.stringify({label:stem,c}));}await b3state.tab.playwright.getByRole('button',{name:/^返回全景$|^Reset view$/}).click();await b3state.tab.playwright.locator('#b3-render-audit[data-camera=overview]').waitFor({state:'attached'});let end=await b3state.cap(stem+'-reset');if(a.background.size!==end.background.size||a.background.position!==end.background.position)throw Error('reset drift');b3state.queue.shift();await b3state.save('final-remaining.json',b3state.queue);}nodeRepl.write({remaining:b3state.queue.length,records:b3state.rows.length});}

save = async(file,v)=>{const tmp=b3dir+file+'.tmp';await fs.writeFile(tmp,JSON.stringify(v,null,2));await fs.rename(tmp,b3dir+file);}

nextPost = async(n)=>{for(let i=0;i<n&&b3state.postB2.length;i++){const q=b3state.postB2[0],stem='b2-post-'+q.scenario.slice(3)+'-'+q.size.width;await b3state.open(q.scenario,q.size);let a=await b3state.cap(stem+'-overview');await b3state.zone(q.scenario.includes('coffee')?'lounge':'meeting');let b=await b3state.cap(stem+'-zone',true);for(const d of [a,b])if(Object.values(b3state.check(d)).some(v=>!v))throw Error(stem);b3state.postB2.shift();}nodeRepl.write({postB2Remaining:b3state.postB2.length});}
